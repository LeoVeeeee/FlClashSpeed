cmake_minimum_required(VERSION 3.20)
project(FlClashSpeedPlugin VERSION 2.0.1 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "FlClashSpeedPlugin requires Windows")
endif()
add_library(FlClashSpeedPlugin SHARED src/Core.cpp src/Runtime.cpp src/Options.cpp src/Plugin.cpp)
target_compile_features(FlClashSpeedPlugin PRIVATE cxx_std_17)
target_compile_definitions(FlClashSpeedPlugin PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(FlClashSpeedPlugin PRIVATE winhttp crypt32 comdlg32 user32 shell32)
set_target_properties(FlClashSpeedPlugin PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(FlClashSpeedPlugin PRIVATE /utf-8 /W4)
  set_property(TARGET FlClashSpeedPlugin PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
